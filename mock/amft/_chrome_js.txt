
(function(){
  // One panel for all three headings, matching the design: whichever you click,
  // the whole inventory opens. Cheaper than three menus and it means a visitor
  // sees everything the site has in one gesture.
  var bar = document.querySelector('.sitenav'),
      panel = document.getElementById('navpanel'),
      btns = document.querySelectorAll('[data-nav]');
  function set(open){
    panel.hidden = !open;
    bar.classList.toggle('open', open);
    btns.forEach(function(b){ b.setAttribute('aria-expanded', open ? 'true' : 'false'); });
  }
  btns.forEach(function(b){
    b.addEventListener('click', function(e){ e.preventDefault(); set(panel.hidden); });
  });
  document.addEventListener('keydown', function(e){ if(e.key === 'Escape') set(false); });
  document.addEventListener('click', function(e){
    if(!panel.hidden && !bar.contains(e.target)) set(false);
  });
})();

(function(){
  // Post the signup in the background instead of letting the browser navigate.
  // Formspree's own "Thanks! The form was submitted successfully. Go back" page
  // is a dead end - it throws the reader off the site mid-thought and gives them
  // a back button as the only way home. Same request, same endpoint, but the
  // answer lands where they were standing.
  function esc(s){ return String(s).replace(/[<>&"]/g, ''); }
  function ok(f){
    var email = (f.querySelector('input[type=email]') || {}).value || '';
    var d = document.createElement('div');
    d.className = 'nlok';
    // Preserve the form type after replaceChild so the analytics success
    // observer can distinguish a contact lead from a newsletter signup.
    d.dataset.kind = f.dataset.kind || 'newsletter';
    d.setAttribute('role', 'status');
    // Two kinds of form share this handler. A message and a mailing-list
    // signup are not the same event and must not get the same answer - being
    // told "you're on the list" after reporting a bug reads as though the
    // report went nowhere.
    if (f.dataset.kind === 'contact'){
      d.innerHTML = '<span class="nlok-tick" aria-hidden="true">&#10003;</span><div>'
        + '<b>Message sent.</b>'
        + '<p>It is read by a person, usually within a day or two'
        + (email ? ', and any reply goes to <b style="display:inline;font-family:inherit;'
            + 'font-size:inherit">' + esc(email) + '</b>' : '')
        + '. If it is a bug, the fix and the date it landed go on the page it affects.</p>'
        + '<p>Got a screenshot? This form cannot carry a file \u2014 send a second message '
        + 'with a link to it, or just describe what looked wrong.</p></div>';
      f.parentNode.replaceChild(d, f);
      return;
    }
    d.innerHTML = '<span class="nlok-tick" aria-hidden="true">&#10003;</span><div>'
      + '<b>You\u2019re on the list.</b>'
      + '<p><b style="display:inline;font-family:inherit;font-size:inherit">'
      + esc(email) + '</b> is on it. The next issue goes out at the start of '
      + 'the month \u2014 a free tool or two, what colleagues are charging, and whatever caught '
      + 'someone out.</p>'
      + '<p>Nothing else happens in the meantime. One click leaves, any time.</p></div>';
    f.parentNode.replaceChild(d, f);
    if (window.gtag) gtag('event', 'newsletter_signup');
  }
  function fail(f){
    var e = f.querySelector('.nlerr');
    if (!e) { e = document.createElement('p'); e.className = 'nlerr'; f.appendChild(e); }
    e.textContent = f.dataset.kind === 'contact'
      ? 'That did not go through, and nothing was sent. Copy your message somewhere safe and '
        + 'try again in a few minutes.'
      : 'That did not go through, and nothing was sent. Check the address and try again.';
  }
  // Both form kinds, or the contact page silently falls back to a native
  // POST and navigates to Formspree's dead-end thanks page - the exact
  // behaviour this handler exists to prevent.
  document.querySelectorAll('form.nlform, form.cform').forEach(function(f){
    f.addEventListener('submit', function(ev){
      ev.preventDefault();
      var btn = f.querySelector('button[type=submit]'), label = btn ? btn.textContent : '';
      if (btn) { btn.setAttribute('aria-busy', 'true'); btn.disabled = true;
                 btn.textContent = 'Sending\u2026'; }
      fetch(f.action, {method: 'POST', body: new FormData(f),
                       headers: {'Accept': 'application/json'}})
        .then(function(r){
          if (r.ok) { ok(f); return; }
          throw new Error('http ' + r.status);
        })
        .catch(function(){
          fail(f);
          if (btn) { btn.removeAttribute('aria-busy'); btn.disabled = false;
                     btn.textContent = label; }
        });
    });
  });
})();
